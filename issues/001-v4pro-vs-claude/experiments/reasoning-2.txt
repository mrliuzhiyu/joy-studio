RK4积分：
```
function lorenzDeriv(x, y, z) {
    return [
        sigma * (y - x),
        x * (rho - z) - y,
        x * y - beta * z
    ];
}

function rk4Step(x, y, z, dt) {
    // k1
    let [dx1, dy1, dz1] = lorenzDeriv(x, y, z);
    // k2
    let [dx2, dy2, dz2] = lorenzDeriv(x + 0.5*dt*dx1, y + 0.5*dt*dy1, z + 0.5*dt*dz1);
    // k3
    let [dx3, dy3, dz3] = lorenzDeriv(x + 0.5*dt*dx2, y + 0.5*dt*dy2, z + 0.5*dt*dz2);
    // k4
    let [dx4, dy4, dz4] = lorenzDeriv(x + dt*dx3, y + dt*dy3, z + dt*dz3);
    
    return [
        x + dt/6 * (dx1 + 2*dx2 + 2*dx3 + dx4),
        y + dt/6 * (dy1 + 2*dy2 + 2*dy3 + dy4),
        z + dt/6 * (dz1 + 2*dz2 + 2*dz3 + dz4)
    ];
}
```
